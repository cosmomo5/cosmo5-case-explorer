    ```typescript
    import gsap from 'gsap';
    import * as THREE from 'three';
    import { computePhyllotaxisPositions } from './phyllotaxis';

    export class ComparisonMode {
      private active = false;
      private gatheredStars: THREE.Mesh[] = [];
      private labels: HTMLElement[] = [];
      private priorState: Map<THREE.Mesh, { opacity: number; starState: string }> = new Map();
      private ctx: gsap.Context;

      constructor(private sceneObjects: SceneObjects) {
        this.ctx = gsap.context(() => {});
      }

      get isActive(): boolean { return this.active; }

      enter(starMeshes: THREE.Mesh[], caseData: Map<string, CaseStudy>): void {
        this.active = true;
        this.gatheredStars = starMeshes;

        // Save prior state of ALL stars for restoration
        for (const star of this.sceneObjects.starMeshes) {
          const mat = star.material as THREE.MeshBasicMaterial;
          this.priorState.set(star, { opacity: mat.opacity, starState: star.userData.starState });
        }

        // Compute formation positions
        const positions = computePhyllotaxisPositions(starMeshes.length);

        // Dim ALL non-gathered stars to 0.08
        this.ctx.add(() => {
          for (const star of this.sceneObjects.starMeshes) {
            if (this.gatheredStars.includes(star)) continue;
            const mat = star.material as THREE.MeshBasicMaterial;
            gsap.to(mat, { opacity: 0.08, duration: 0.4 });
          }
        });

        // Fly gathered stars to formation
        this.ctx.add(() => {
          starMeshes.forEach((star, i) => {
            const target = positions[i];
            gsap.to(star.position, {
              x: target.x, y: target.y, z: target.z,
              duration: 0.9, ease: 'power2.inOut',
              delay: i * 0.05 // slight stagger
            });
          });
        });

        // Add DM Mono labels (DOM overlay)
        // Create label elements positioned near each star

        // Hide UI elements
        document.body.classList.add('comparison-active');
      }

      exit(): void {
        if (!this.active) return;
        this.active = false;

        // Kill all GSAP tweens in context
        this.ctx.revert();

        // Return stars to track positions (not orbit — they're still in the reading list)
        // Restore all star opacities from priorState
        for (const [star, state] of this.priorState) {
          const mat = star.material as THREE.MeshBasicMaterial;
          gsap.to(mat, { opacity: state.opacity, duration: 0.3 });
          star.userData.starState = state.starState;
        }

        // Remove labels
        this.labels.forEach(l => l.remove());
        this.labels = [];

        // Restore UI
        document.body.classList.remove('comparison-active');
        this.priorState.clear();
      }
    }
    ```

    Adapt based on actual codebase structure. The key is: save state on enter, restore on exit.
