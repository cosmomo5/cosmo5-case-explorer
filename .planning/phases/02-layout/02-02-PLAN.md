**Implementation** — rewrite `src/scene/renderer.ts`. Key changes only, preserve all controls config:

```typescript
export function createRenderer(
  canvas: HTMLCanvasElement,
  container: HTMLElement,   // NEW — was only canvas
): RendererContext {
  const renderer = new THREE.WebGLRenderer({ canvas, antialias: true, alpha: true });
  renderer.setPixelRatio(Math.min(devicePixelRatio, 2));
  // Do NOT call renderer.setSize here — onContainerResize() will handle it

  const scene = new THREE.Scene();
  const camera = new THREE.PerspectiveCamera(60, 1, 0.1, 3000);
  camera.position.set(80, 880, 280);

  // ... all controls config unchanged ...

  const onContainerResize = () => {
    const w = container.clientWidth;
    const h = container.clientHeight;
    camera.aspect = w / h;
    camera.updateProjectionMatrix();
    renderer.setSize(w, h, false);  // false = don't override canvas CSS dimensions
  };

  const resizeObserver = new ResizeObserver(onContainerResize);
  resizeObserver.observe(container);
  onContainerResize();  // initialize synchronously — prevents 300x150 flash

  return {
    scene, camera, renderer, controls,
    dispose: () => {
      resizeObserver.disconnect();
      controls.dispose();
      renderer.dispose();
    },
  };
}
```

Remove the old `addEventListener('resize', onResize)` line entirely. No `removeEventListener` in dispose either.
