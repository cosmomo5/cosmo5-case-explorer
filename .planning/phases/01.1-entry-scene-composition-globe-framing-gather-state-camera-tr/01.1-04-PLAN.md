```typescript
  // (0) Clear Entry camera offset over the 1.2s window (parallel with sequences 1–5)
  // This runs in parallel with the planet fade and star release below.
  const camera = objects.camera as THREE.PerspectiveCamera;
  const hasViewOffset = camera.view?.enabled;
  
  if (hasViewOffset) {
    const containerEl = document.getElementById('scene-container');
    const W = containerEl?.clientWidth ?? window.innerWidth;
    const H = containerEl?.clientHeight ?? window.innerHeight;
    
    // Read current offsets or default to Entry values
    const currentOffsetX = camera.view?.offsetX ?? 0;
    const currentOffsetY = camera.view?.offsetY ?? 0;
    
    const proxy = { dx: currentOffsetX, dy: currentOffsetY };
    
    gsap.to(proxy, {
      dx: 0, dy: 0,
      duration: 1.2 * s,
      ease: 'power1.inOut',
      onUpdate: () => {
        camera.setViewOffset(W, H, proxy.dx, proxy.dy, W, H);
      },
      onComplete: () => {
        camera.clearViewOffset();
      },
    });
  }

  // (1) Dim planet core (existing sequence)
  // ... rest of function continues unchanged
```
