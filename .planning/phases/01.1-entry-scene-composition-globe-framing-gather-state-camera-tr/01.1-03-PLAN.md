Extended with opacity reveal:
```typescript
const prefersReduced = window.matchMedia?.('(prefers-reduced-motion: reduce)').matches;
const positionDuration = prefersReduced ? 0.2 : 1.2; // existing scaling
const opacityDuration = prefersReduced ? 0.2 : 0.8;

for (const star of objects.starMeshes) {
  if (star.userData.isPlanetaryTarget) {
    const { x, y, z } = computeOrbit(...);
    
    // Existing: position tween
    gsap.to(star.position, {
      x, y, z,
      duration: positionDuration,
      ease: 'power2.out',
      onComplete: ...
    });
    
    // NEW: visibility and opacity reveal (D-01.1-03)
    // Set visible BEFORE tween starts (otherwise tween won't render anything)
    star.visible = true;
    const sMat = star.material as THREE.MeshBasicMaterial;
    gsap.to(sMat, {
      opacity: 0.85,
      duration: opacityDuration,
      delay: 0.15,  // slight delay so opacity starts mid-position-tween
      ease: 'power2.out',
    });
    
    // Reveal glow sprite opacity as well
    const glowSp = star.userData.glowSprite as THREE.Sprite | undefined;
    if (glowSp) {
      const spMat = glowSp.material as THREE.SpriteMaterial;
      const baseOpacity = (glowSp.userData.baseOpacity as number) ?? 0.65;
      gsap.to(spMat, {
        opacity: baseOpacity,
        duration: opacityDuration,
        delay: 0.15,
        ease: 'power2.out',
      });
    }
  }
}
```

The key points:
1. `star.visible = true` must be set BEFORE the opacity tween starts (so the mesh renders)
2. Opacity tweens from 0 (set during boot in Task 01.1-04) to 0.85 / baseOpacity
3. Duration scales with prefersReduced factor
4. Delay of 0.15s means opacity fades in while the star is already migrating (nice visual overlap)
