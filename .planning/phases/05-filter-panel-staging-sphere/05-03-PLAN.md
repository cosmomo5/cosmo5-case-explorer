         const matched = objects.allCases.filter(c => matchesFilter(c, f));
         if (objects.mode === 'global') {
           if (f.isActive() && matched.length > 0) {
             if (!objects.nucleusMeshes.has(STAGING_KEY)) {
               createStagingNucleus(objects, matched);
               applyStagingPush(objects);
               zoomTowardStaging(objects);
             } else {
               // filter changed — update star positions without recreating sphere
               animateStarsToStaging(objects, matched);
             }
           } else {
             if (objects.nucleusMeshes.has(STAGING_KEY)) {
               returnStarsFromStaging(objects);
               // Delay disposal until return tweens complete (~700ms).
