    ```typescript
    else if (objects.mode === 'focused') {
      const isActive = f.isActive();
      for (const [id, mesh] of objects.starMeshes) {
        const caseStudy = objects.allCases.find(c => c.id === id);
        if (!caseStudy) continue;

        const mat = mesh.material as THREE.MeshStandardMaterial;
        const currentState: StarState = mesh.userData.starState ?? 'idle';

        // Precedence: only apply filterDim if star is idle or filterDim
        // Higher states (reading, selected, ghost) are unaffected
        if (currentState === 'idle' || currentState === 'filterDim') {
          const matches = !isActive || matchesFilter(caseStudy, f);
          const targetOpacity = matches ? 1.0 : 0.12;
          const targetState: StarState = matches ? 'idle' : 'filterDim';

          gsap.killTweensOf(mat);
          gsap.to(mat, {
            opacity: targetOpacity,
            duration: 0.2,
            ease: 'power2.out',
            onComplete: () => { mesh.userData.starState = targetState; },
          });
          mesh.userData.starState = targetState;
        }
      }
    }
    ```
