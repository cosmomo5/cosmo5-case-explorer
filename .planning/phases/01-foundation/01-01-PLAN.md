    ```typescript
    export type ViewState =
      | { kind: 'global' }
      | { kind: 'constellation'; nucleusKey: string }
      | { kind: 'reading'; nucleusKey: string; caseId: string };

    export type ViewEvent =
      | { type: 'ENTER_CONSTELLATION'; nucleusKey: string }
      | { type: 'EXIT_CONSTELLATION' }
      | { type: 'OPEN_READING'; caseId: string }
      | { type: 'CLOSE_READING' };

    export type ViewStateListener = (next: ViewState, prev: ViewState, event: ViewEvent) => void;

    export interface ViewStateMachine {
      getState(): ViewState;
      dispatch(event: ViewEvent): void;
      subscribe(listener: ViewStateListener): () => void;
    }

    export function createViewStateMachine(initial?: ViewState): ViewStateMachine;
    ```

    Transition table (throw on all invalid transitions):
    - global + ENTER_CONSTELLATION → constellation
    - constellation + EXIT_CONSTELLATION → global
    - any + OPEN_READING → throw "Transition OPEN_READING not wired until Phase 5"
    - any + CLOSE_READING → throw "Transition CLOSE_READING not wired until Phase 5"
    - any other combination → throw with descriptive message

    Implementation uses a `Set<ViewStateListener>` and calls all listeners synchronously inside the `transition()` helper.

    Do NOT import Three.js, GSAP, or any DOM API. This module is pure TypeScript, runnable in Node.

    ```typescript
      nucleusKey: string;
    }

      remove(caseId: string): void;
      clear(): void;

      subscribe(listener: ShortlistListener): () => void;
    ```
