```typescript
// @vitest-environment jsdom
import { describe, it, expect, vi, beforeEach, afterEach } from 'vitest';
import { mountFilterPanel } from './filter-panel';
import { createFilterState } from '../data/filter';
import { viewState } from '../data/view-state';
import type { SceneObjects } from '../scene/scene-state';
import type { CaseStudy } from '../data/types';

// Fixture: mock cases
const mockCases: CaseStudy[] = [
  { id: 'c1', title: 'Fintech Growth', service: 'Performance Marketing', region: 'EMEA', year: 2024, permission: true },
  { id: 'c2', title: 'Brand Refresh', service: 'Brand', region: 'APAC', year: 2023, permission: true },
  // ... more cases
];

// Fixture: mock SceneObjects
function makeObjects(): SceneObjects {
  return {
    starMeshes: [],
    nucleusMeshes: new Map(),
    nucleusLabels: new Map(),
    filterState: createFilterState(),
    // ... other required fields
  };
}

describe('Unified Left Column', () => {
  let container: HTMLElement;
  let objects: SceneObjects;

  beforeEach(() => {
    container = document.createElement('aside');
    container.id = 'filters';
    document.body.appendChild(container);
    objects = makeObjects();
  });

  afterEach(() => {
    document.body.removeChild(container);
  });

  describe('Input behavior', () => {
    it('should update readout when input text changes', () => {
      const onFilterChange = vi.fn();
      mountFilterPanel(container, objects, vi.fn(), undefined, mockCases, onFilterChange);

      const input = container.querySelector('input[type="text"]') as HTMLInputElement;
      const readout = container.querySelector('.filters-readout') as HTMLElement;

      input.value = 'fintech';
      input.dispatchEvent(new Event('input'));

      // After debounce, readout should show match count
      setTimeout(() => {
        expect(readout.textContent).toMatch(/\d+ CASES/);
      }, 100);
    });

    it('should show empty-state hint when input is empty', () => {
      mountFilterPanel(container, objects, vi.fn(), undefined, mockCases);

      const input = container.querySelector('input[type="text"]') as HTMLInputElement;
      const hint = container.querySelector('.query-hint') as HTMLElement;

      input.value = '';
      input.dispatchEvent(new Event('input'));

      expect(hint.classList.contains('visible')).toBe(true);
      expect(hint.textContent).toContain('Or explore the full catalog');
    });

    it('should show no-match hint when query matches no cases', () => {
      mountFilterPanel(container, objects, vi.fn(), undefined, mockCases);

      const input = container.querySelector('input[type="text"]') as HTMLInputElement;
      const hint = container.querySelector('.query-hint') as HTMLElement;

      input.value = 'xyzunknownquery';
      input.dispatchEvent(new Event('input'));

      expect(hint.classList.contains('visible')).toBe(true);
      expect(hint.textContent).toContain('No observations match');
    });
  });

  describe('Keyboard handlers', () => {
    it('should trigger wave on Enter key', () => {
      const onFilterChange = vi.fn();
      mountFilterPanel(container, objects, vi.fn(), undefined, mockCases, onFilterChange);

      const input = container.querySelector('input[type="text"]') as HTMLInputElement;
      input.value = 'fintech';

      const enterEvent = new KeyboardEvent('keydown', { key: 'Enter' });
      input.dispatchEvent(enterEvent);

      expect(onFilterChange).toHaveBeenCalled();
    });

    it('should clear input on Escape key', () => {
      mountFilterPanel(container, objects, vi.fn(), undefined, mockCases);

      const input = container.querySelector('input[type="text"]') as HTMLInputElement;
      input.value = 'fintech';

      const escapeEvent = new KeyboardEvent('keydown', { key: 'Escape' });
      input.dispatchEvent(escapeEvent);

      expect(input.value).toBe('');
    });
  });

  describe('Query history', () => {
    it('should display and populate input on chip click', () => {
      mountFilterPanel(container, objects, vi.fn(), undefined, mockCases);

      const input = container.querySelector('input[type="text"]') as HTMLInputElement;
      const historyContainer = container.querySelector('.query-history') as HTMLElement;

      // History should contain at least one chip from loadQueryHistory
      const chip = historyContainer.querySelector('.query-chip') as HTMLButtonElement;
      if (chip) {
        chip.click();
        const chipText = chip.textContent || '';
        expect(input.value).toBe(chipText);
      }
    });
  });

  describe('View-state auto-expand', () => {
    it('should remove .collapsed on ENTER_CONSTELLATION', () => {
      container.classList.add('collapsed');
      mountFilterPanel(container, objects, vi.fn(), undefined, mockCases);

      viewState.dispatch({ type: 'ENTER_CONSTELLATION', nucleusKey: 'vertical:tech' });

      expect(container.classList.contains('collapsed')).toBe(false);
    });

    it('should add .collapsed on EXIT_CONSTELLATION', () => {
      mountFilterPanel(container, objects, vi.fn(), undefined, mockCases);

      viewState.dispatch({ type: 'EXIT_CONSTELLATION' });

      expect(container.classList.contains('collapsed')).toBe(true);
    });
  });
});
```
