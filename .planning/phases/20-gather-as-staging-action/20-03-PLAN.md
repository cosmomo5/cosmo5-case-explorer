```typescript
/**
 * Determine the Gather button's current state based on filter activity and sphere presence.
 * Returns the state name, matched case count, and staged case count for label rendering.
 */
function determineGatherButtonState(
  objects: SceneObjects,
  allCases: EnrichedCaseStudy[]
): { 
  state: 'disabled' | 'armed' | 'staged-current' | 'staged-drifted' | 'stranded'; 
  matchedCount: number; 
  stagedCount: number 
} {
  const isActive = objects.filterState.isActive();
  const matchedCount = isActive ? countMatches(objects, allCases) : 0;
  const sphereExists = objects.nucleusMeshes.has(STAGING_KEY);
  const stagedCount = sphereExists 
    ? (objects.nucleusOrbits[STAGING_KEY]?.cases?.length ?? 0)
    : 0;

  // State tree per UI-SPEC Decision Tree (lines 141-152)
  if (!isActive || matchedCount === 0) {
    return { state: 'disabled', matchedCount, stagedCount };
  }
  if (!sphereExists) {
    return { state: 'armed', matchedCount, stagedCount };
  }
  if (!isActive && sphereExists) {
    return { state: 'stranded', matchedCount, stagedCount };
  }
  if (matchedCount === stagedCount) {
    return { state: 'staged-current', matchedCount, stagedCount };
  }
  return { state: 'staged-drifted', matchedCount, stagedCount };
}

/**
 * Render the Gather button label based on current state.
 * Matches UI-SPEC Copy Specification (lines 456-466).
 */
function renderGatherButtonLabel(
  state: 'disabled' | 'armed' | 'staged-current' | 'staged-drifted' | 'stranded',
  isActive: boolean,
  matchedCount: number,
  stagedCount: number
): string {
  switch (state) {
    case 'disabled':
      return isActive && matchedCount === 0 ? '0 CASES — REFINE' : 'GATHER';
    case 'armed':
      return 'GATHER';
    case 'staged-current':
      return `GATHERED · ${matchedCount}`;
    case 'staged-drifted':
      return `RE-GATHER · ${stagedCount}→${matchedCount}`;
    case 'stranded':
      return 'DISMISS';
  }
}

/**
 * Render aria-label for accessibility (per UI-SPEC Accessibility Checklist).
 */
function renderGatherButtonAriaLabel(
  state: 'disabled' | 'armed' | 'staged-current' | 'staged-drifted' | 'stranded',
  matchedCount: number,
  stagedCount: number
): string {
  switch (state) {
    case 'disabled':
      return `Cannot gather: no filters active or empty result set`;
    case 'armed':
      return `Gather ${matchedCount} cases to staging area`;
    case 'staged-current':
      return `Gathered: ${matchedCount} cases currently staged`;
    case 'staged-drifted':
      return `Filter changed. Re-gather with new set: ${stagedCount} to ${matchedCount} cases`;
    case 'stranded':
      return `Dismiss orphaned staging sphere`;
  }
}
```

/**
 * Helper function to recompute and update the Gather button's visuals after state change.
 * Called after gather() or disposeStagingNucleus() to reflect new sphere state.
 */
function recomputeGatherButtonUI(): void {
  const gatherBtn = container.querySelector('.gather-button') as HTMLButtonElement | null;
  if (!gatherBtn) return;
  
  const newState = determineGatherButtonState(objects, allCases);
  const newLabel = renderGatherButtonLabel(
    newState.state,
    objects.filterState.isActive(),
    newState.matchedCount,
    newState.stagedCount
  );
  const newAriaLabel = renderGatherButtonAriaLabel(
    newState.state,
    newState.matchedCount,
    newState.stagedCount
  );
  
  // Update class names
  gatherBtn.className = [
    'gather-button',
    newState.state === 'disabled' ? 'disabled' : '',
    newState.state === 'staged-current' ? 'staged-current' : '',
    newState.state === 'staged-drifted' ? 'staged-drifted' : '',
    newState.state === 'stranded' ? 'stranded' : '',
  ]
    .filter(Boolean)
    .join(' ');
  
  // Update button content and attributes
  gatherBtn.textContent = newLabel;
  gatherBtn.setAttribute('aria-label', newAriaLabel);
  gatherBtn.setAttribute('aria-pressed', newState.state.startsWith('staged') ? 'true' : 'false');
  
  // Update disabled state
  if (newState.state === 'disabled') {
    gatherBtn.setAttribute('disabled', '');
  } else {
    gatherBtn.removeAttribute('disabled');
  }
}
