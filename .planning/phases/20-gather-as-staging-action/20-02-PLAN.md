```typescript
export function gather(objects: SceneObjects): void {
  // Compute matched cases from current filter
  const matchedCases = objects.starMeshes
    .map(starMesh => starMesh.userData.caseStudy as EnrichedCaseStudy)
    .filter(caseStudy => matchesFilter(caseStudy, objects.filterState));
  
  // If no matches, don't create sphere (button should be disabled anyway)
  if (matchedCases.length === 0) {
    return;
  }
  
  // Decide: create new sphere or update existing
  const sphereExists = objects.nucleusMeshes.has(STAGING_KEY);
  
  if (!sphereExists) {
    // First gather: create, push nuclei outward, zoom camera toward sphere
    createStagingNucleus(objects, matchedCases);
    applyStagingPush(objects);
    zoomTowardStaging(objects);
  } else {
    // Sphere already present: animate matched stars to it (replaces existing set)
    animateStarsToStaging(objects, matchedCases);
  }
}
```
