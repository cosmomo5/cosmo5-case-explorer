```ts
import * as THREE from 'three';
import { gsap } from 'gsap';
import type { SceneObjects } from './scene-state';
import { starPositionAt } from './orbitals/star-orbit';

export interface Vec3 { x: number; y: number; z: number; }

/**
 * Fibonacci sphere — distributes N points evenly on the surface of a unit sphere.
 * Classic golden-angle + z-linear formulation; yields visually even coverage
 * without clumping at poles.
 */
export function fibonacciSphere(count: number, radius = 1): Vec3[] {
  const points: Vec3[] = [];
  const goldenAngle = Math.PI * (3 - Math.sqrt(5));
  for (let i = 0; i < count; i++) {
    const z = 1 - (2 * i) / Math.max(count - 1, 1);
    const r = Math.sqrt(1 - z * z);
    const theta = goldenAngle * i;
    points.push({ x: Math.cos(theta) * r * radius, y: z * radius, z: Math.sin(theta) * r * radius });
  }
  return points;
}

/**
 * Place all case-stars on the planet surface. Instant (no animation).
 * Tags each star with isPlanetaryTarget=true so tickOrbits will skip them.
 * Stores the star's original orbital home in userData.planetaryReleaseTarget
 * for releaseStarsToHomes to use.
 */
export function gatherStarsToPlanet(
  objects: SceneObjects,
  planetCenter: Vec3,
  surfaceRadius: number,
): void {
  const positions = fibonacciSphere(objects.starMeshes.length, surfaceRadius);
  objects.starMeshes.forEach((star, i) => {
    // Release target = the current orbital home (recompute from starOrbit at t=0)
    const orbit = star.userData.starOrbit as any;
    const home = starPositionAt(orbit, 0);
    star.userData.planetaryReleaseTarget = new THREE.Vector3(home.x, home.y, home.z);
    // Set surface position
    const p = positions[i];
    star.position.set(planetCenter.x + p.x, planetCenter.y + p.y, planetCenter.z + p.z);
    star.userData.isPlanetaryTarget = true;
  });
}

/**
 * Tween every planetary-target star from its current (surface) position to its
 * orbital home. Clears the flag on completion so tickOrbits resumes control.
 * Respects prefers-reduced-motion: uses 0.2s duration instead of 1.2s.
 */
export function releaseStarsToHomes(objects: SceneObjects): Promise<void> {
  const prefersReduced = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  const duration = prefersReduced ? 0.2 : 1.2;

  const pending: Promise<void>[] = [];
  for (const star of objects.starMeshes) {
    if (!star.userData.isPlanetaryTarget) continue;
    const target = star.userData.planetaryReleaseTarget as THREE.Vector3 | undefined;
    if (!target) { star.userData.isPlanetaryTarget = false; continue; }
    pending.push(new Promise<void>((resolve) => {
      gsap.killTweensOf(star.position);
      gsap.to(star.position, {
        x: target.x, y: target.y, z: target.z,
        duration,
        ease: 'power2.out',
        onComplete: () => {
          star.userData.isPlanetaryTarget = false;
          delete star.userData.planetaryReleaseTarget;
          resolve();
        },
      });
    }));
  }
  return Promise.all(pending).then(() => {});
}
```

**File 2: src/scene/planet.ts**

```ts
import * as THREE from 'three';

export const PLANET_RADIUS = 30;      // world units — visually prominent sphere
export const PLANET_RING_RADIUS = 44; // outer ring
const TEAL = 0x5ecfc0;

export interface PlanetGroup {
  group: THREE.Group;
  sphere: THREE.Mesh;
  ring: THREE.Mesh;
}

export function createPlanet(scene: THREE.Scene): PlanetGroup {
  const group = new THREE.Group();
  group.name = 'planetary-overture-planet';

  // Sphere: ShaderMaterial or MeshStandardMaterial with teal emissive glow
  const sphereGeo = new THREE.SphereGeometry(PLANET_RADIUS, 48, 48);
  const sphereMat = new THREE.MeshStandardMaterial({
    color: 0x0b0c10,
    emissive: TEAL,
    emissiveIntensity: 0.12,
    roughness: 0.65,
    metalness: 0.2,
    transparent: true,
    opacity: 1,
  });
  const sphere = new THREE.Mesh(sphereGeo, sphereMat);
  group.add(sphere);

  // Ring: torus, oriented to tilt slightly toward camera (rotation X = 72deg)
  const ringGeo = new THREE.TorusGeometry(PLANET_RING_RADIUS, 0.15, 8, 128);
  const ringMat = new THREE.MeshBasicMaterial({
    color: TEAL,
    transparent: true,
    opacity: 0.12,
  });
  const ring = new THREE.Mesh(ringGeo, ringMat);
  ring.rotation.x = Math.PI / 2 - (18 * Math.PI / 180); // tilt ~18deg from horizontal
  group.add(ring);

  scene.add(group);
  return { group, sphere, ring };
}

export function disposePlanet(planet: PlanetGroup, scene: THREE.Scene): void {
  scene.remove(planet.group);
  planet.sphere.geometry.dispose();
  (planet.sphere.material as THREE.Material).dispose();
  planet.ring.geometry.dispose();
  (planet.ring.material as THREE.Material).dispose();
}
```

```ts
async function choreographPlanetaryToGlobal(
  objects: SceneObjects,
  planet: PlanetGroup,
): Promise<void> {
  const prefersReduced = window.matchMedia('(prefers-reduced-motion: reduce)').matches;
  const s = prefersReduced ? 0.33 : 1.0; // scale factor for durations

  // (1) Dim planet core
  gsap.to((planet.sphere.material as THREE.MeshStandardMaterial), {
    emissiveIntensity: 0.02,
    opacity: 0.35,
    duration: 0.4 * s,
    ease: 'power1.out',
  });

  // (2) Start release with a 0.2s lead after dim begins
  setTimeout(() => {
    releaseStarsToHomes(objects);
  }, 200 * s);

  // (3) Ring fade
  gsap.to((planet.ring.material as THREE.MeshBasicMaterial), {
    opacity: 0,
    duration: 0.6 * s,
    ease: 'power1.out',
    delay: 0.4 * s,
  });

  // (4) Planet sphere fade to 0 starting at t=1.0s
  const fadePromise = new Promise<void>((resolve) => {
    gsap.to((planet.sphere.material as THREE.MeshStandardMaterial), {
      opacity: 0,
      duration: 0.6 * s,
      delay: 1.0 * s,
      ease: 'power1.in',
      onComplete: () => resolve(),
    });
  });

  await fadePromise;
  disposePlanet(planet, objects.scene);

  // (5) Fade cluster labels in
  for (const [, div] of objects.nucleusLabels.entries()) {
    div.classList.add('show');
  }
}
```
