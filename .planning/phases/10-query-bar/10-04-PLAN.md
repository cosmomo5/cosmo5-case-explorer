  <action>
    Create src/ui/query-bar.ts with the following exports:

    ```typescript
    import { applyQuery } from '../data/keyword-map';
    import { matchesFilter, createFilterState, FilterState } from '../data/filter';
    import { applyFilterWave } from '../scene/scene-state';
    import { applyFilter } from '../scene/scene-state';
    import type { SceneObjects } from '../scene/scene-state'; // or wherever SceneObjects is
    import type { CaseStudy } from '../data/types';

    interface QueryBarDeps {
      sceneObjects: SceneObjects;
      filterState: FilterState;
      cases: CaseStudy[];
      onFilterChange: () => void; // callback for any external state sync
    }

    export function countMatches(objects: SceneObjects, filterState: FilterState): number {
      return objects.starMeshes.filter(star =>
        matchesFilter(star.userData.caseStudy as CaseStudy, filterState)
      ).length;
    }

    export function countMatchingClusters(objects: SceneObjects, filterState: FilterState): number {
      return Object.values(objects.nucleusOrbits).filter(nucOrbit =>
        nucOrbit.cases.some(c => matchesFilter(c, filterState))
      ).length;
    }

    export function mountQueryBar(deps: QueryBarDeps): void {
      const input = document.getElementById('queryInput') as HTMLInputElement;
      const readout = document.getElementById('queryReadout')!;
      const clearBtn = document.getElementById('queryClear')!;
      if (!input || !readout || !clearBtn) return;

      // Initial readout
      updateReadout(deps, readout);

      // Debounce for wave animation
      let waveTimer: ReturnType<typeof setTimeout>;

      function handleInput(): void {
        const q = input.value;

        // Show/hide clear button
        clearBtn.classList.toggle('visible', q.length > 0);

        // Apply query to filterState (synchronous)
        applyQuery(q, deps.filterState, deps.cases);

        // Immediate readout update
        updateReadout(deps, readout);

        // Debounced wave animation
        clearTimeout(waveTimer);
        waveTimer = setTimeout(() => {
          if (q.length > 0) {
            applyFilterWave(deps.sceneObjects, deps.filterState);
          } else {
            // Empty query: restore idle state
            applyFilter(deps.sceneObjects, createFilterState());
          }
          deps.onFilterChange();
        }, 80);
      }

      input.addEventListener('input', handleInput);

      // Enter submits immediately (skip debounce)
      input.addEventListener('keydown', (e: KeyboardEvent) => {
        if (e.key === 'Enter') {
          clearTimeout(waveTimer);
          const q = input.value;
          applyQuery(q, deps.filterState, deps.cases);
          updateReadout(deps, readout);
          if (q.length > 0) {
            applyFilterWave(deps.sceneObjects, deps.filterState);
          } else {
            applyFilter(deps.sceneObjects, createFilterState());
          }
          deps.onFilterChange();
        }
        if (e.key === 'Escape') {
          input.value = '';
          handleInput();
          input.blur();
        }
      });

      // Clear button
      clearBtn.addEventListener('click', () => {
        input.value = '';
        handleInput();
        input.focus();
      });
    }

    function updateReadout(deps: QueryBarDeps, readout: HTMLElement): void {
      const n = countMatches(deps.sceneObjects, deps.filterState);
      const m = countMatchingClusters(deps.sceneObjects, deps.filterState);
      readout.textContent = `${n} CASES · ${m} CLUSTERS`;
    }
    ```

    Adapt imports based on actual module paths found in the codebase. Read scene-state.ts to find the exact SceneObjects type export path.
