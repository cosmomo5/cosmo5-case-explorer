       ```typescript
       export function applyQuery(q: string, fs: FilterState, cases: CaseStudy[]): void {
         const norm = q.trim().toLowerCase();

         // Clear all dimension sets
         fs.countries = new Set();
         fs.services = new Set();
         fs.years = new Set();
         fs.nucleusKeys = new Set();
         fs.proofTypes = new Set();
         fs.impactBands = new Set();
         delete fs.searchText;

         if (!norm) return; // empty query = clear all

         let anyHintMatched = false;
         for (const hint of KEYWORD_HINTS) {
           if (hint.keywords.some(kw => norm.includes(kw))) {
             hint.apply(fs);
             anyHintMatched = true;
           }
         }

         // Region expansion: if region hints added markers, resolve to country names
         // (implementation detail — collect countries from cases where region matches)

         if (!anyHintMatched && norm.length > 0) {
           fs.searchText = norm;
         }
       }
       ```
