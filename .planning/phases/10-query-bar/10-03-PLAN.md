    ```typescript
    export function applyFilterWave(objects: SceneObjects, filterState: FilterState): void {
      // 1. Collect positions of matching cluster nuclei
      const matchingNucleiPos: THREE.Vector3[] = [];
      for (const nucOrbit of Object.values(objects.nucleusOrbits)) {
        const hasMatch = nucOrbit.cases.some(c => matchesFilter(c, filterState));
        if (hasMatch) matchingNucleiPos.push(nucOrbit.nucleus.position.clone());
      }

      // 2. Sort stars by minimum distance to nearest matching nucleus
      const sorted = [...objects.starMeshes].sort((a, b) => {
        const distA = matchingNucleiPos.length > 0
          ? Math.min(...matchingNucleiPos.map(np => a.position.distanceTo(np)))
          : 0;
        const distB = matchingNucleiPos.length > 0
          ? Math.min(...matchingNucleiPos.map(np => b.position.distanceTo(np)))
          : 0;
        return distA - distB;
      });

      // 3. Staggered GSAP: ~600ms total
      const TOTAL_S = 0.6;
      const count = sorted.length;

      sorted.forEach((star, i) => {
        const state = star.userData.starState as StarState;
        // Respect precedence: skip reading/selected/ghost
        if (state === 'reading' || state === 'selected' || state === 'ghost') return;

        const c = star.userData.caseStudy as CaseStudy;
        const matches = matchesFilter(c, filterState);
        const targetOpacity = matches ? 0.85 : 0.12;

        star.userData.starState = matches ? 'filterMatch' : 'filterDim';

        const mat = star.material as THREE.MeshBasicMaterial;
        // Kill any in-progress tween on this material before starting a new one
        gsap.killTweensOf(mat);
        const delay = count > 1 ? (i / (count - 1)) * TOTAL_S : 0;
        gsap.to(mat, { opacity: targetOpacity, duration: 0.15, delay, ease: 'power2.out' });
      });
    }
    ```

    Ensure imports: gsap, THREE, matchesFilter, CaseStudy, StarState are available (most should already be imported in scene-state.ts).
