    ```typescript
    export function buildRingLine(
      radius: number,
      center: THREE.Vector3,
      planeNormal: THREE.Vector3,
      color: number,
    ): THREE.LineLoop {
      const segments = 128;
      const pts: THREE.Vector3[] = [];
      const q = new THREE.Quaternion().setFromUnitVectors(
        new THREE.Vector3(0, 0, 1),
        planeNormal.clone().normalize(),
      );
      for (let i = 0; i < segments; i++) {
        const a = (i / segments) * Math.PI * 2;
        const local = new THREE.Vector3(radius * Math.cos(a), radius * Math.sin(a), 0);
        pts.push(local.applyQuaternion(q).add(center));
      }
      const geo = new THREE.BufferGeometry().setFromPoints(pts);
      const mat = new THREE.LineBasicMaterial({
        color,
        transparent: true,
        opacity: 0.35,
        depthWrite: false,
      });
      return new THREE.LineLoop(geo, mat);
    }

    export function disposeRingLines(lines: THREE.LineLoop[], scene: THREE.Scene): void {
      for (const line of lines) {
        scene.remove(line);
        line.geometry.dispose();
        (line.material as THREE.LineBasicMaterial).dispose();
      }
    }
    ```

    ```typescript
      // --- Ring orbit computation (ORB-01/02) ---
      const enrichedCases = nucleusOrbit.cases as EnrichedCaseStudy[];
      const autoDim = autoSelectRingDimension(enrichedCases, objects.groupMode);
      objects.currentRingDimension = autoDim;

      const config = {
        baseRadius: 40,
        ringSpacing: Math.max(20, (nucleus.userData.starRMax - 40) / Math.max(1, (() => {
          const vals = new Set(enrichedCases.map(c => {
            if (autoDim === 'proofType') return c.proofType;
            if (autoDim === 'impactBand') return c.impactScale;
            if (autoDim === 'year') return String(c.year);
            if (autoDim === 'country') return c.countries[0] ?? 'Other';
            return c.region ?? 'Other';
          }));
          return Math.min(8, Math.max(1, vals.size));
        })())),
        angularSpeedBase: 0.00003,
      };

      const ringOrbits = computeConcentricOrbits(enrichedCases, autoDim, nucleusOrbit, config);

      // Assign ringOrbit to each star in this nucleus
      for (const star of objects.starMeshes) {
        if (star.userData.starOrbit?.nucleusKey !== key) continue;
        const caseId = (star.userData.caseStudy as CaseStudy).id;
        const ro = ringOrbits.find(r => r.caseStudy.id === caseId);
        if (ro) {
          star.userData.ringOrbit = ro;
          star.userData.isStagingTarget = false; // tick loop will drive position
        }
      }

      // --- Ring lines (ORB-04) ---
      // Dispose any existing ring lines first (prevents accumulation on rapid cycling)
      disposeRingLines(objects.ringLineMeshes, scene);
      objects.ringLineMeshes = [];
      objects.ringLineValues = [];

      const nucCenter = new THREE.Vector3(nucleus.position.x, nucleus.position.y, nucleus.position.z);
      const u = new THREE.Vector3(nucleusOrbit.orbitU.x, nucleusOrbit.orbitU.y, nucleusOrbit.orbitU.z);
      const v = new THREE.Vector3(nucleusOrbit.orbitV.x, nucleusOrbit.orbitV.y, nucleusOrbit.orbitV.z);
      const planeNormal = new THREE.Vector3().crossVectors(u, v).normalize();

      // Group ring orbits by ringValue to find unique radii
      const seenRadii = new Map<string, number>(); // ringValue → radius
      for (const ro of ringOrbits) {
        seenRadii.set(ro.ringValue, ro.ringRadius);
      }

      for (const [ringValue, radius] of seenRadii.entries()) {
        const line = buildRingLine(radius, nucCenter, planeNormal, color);
        scene.add(line);
        objects.ringLineMeshes.push(line);
        objects.ringLineValues.push(ringValue);

      const EXPANSION_MULTIPLIER = 4.5;
      for (const [k, mesh] of objects.nucleusMeshes.entries()) {
        if (k === key) continue;
        const home = mesh.userData.homePos as THREE.Vector3;
        if (!home) continue;
        const expandedPos = home.clone().multiplyScalar(EXPANSION_MULTIPLIER);

        gsap.to(mesh.position, { x: expandedPos.x, y: expandedPos.y, z: expandedPos.z,
          duration: 1.0, ease: 'power2.inOut' });
        if (mesh.userData.halo) {
          gsap.to((mesh.userData.halo as THREE.Mesh).position,
            { x: expandedPos.x, y: expandedPos.y, z: expandedPos.z, duration: 1.0, ease: 'power2.inOut' });
        }
        // Stars of non-focused nuclei move with their nucleus — isStagingTarget prevents tick fight
        for (const star of objects.starMeshes) {
          if ((star.userData.starOrbit as StarOrbit)?.nucleusKey !== k) continue;
          star.userData.isStagingTarget = true;
          const offset = star.position.clone().sub(home);
          const newStarPos = expandedPos.clone().add(offset);
          gsap.to(star.position, { x: newStarPos.x, y: newStarPos.y, z: newStarPos.z,
            duration: 1.0, ease: 'power2.inOut' });
          // Fade out stars of non-focused nuclei
          gsap.to((star.material as THREE.MeshBasicMaterial), { opacity: 0.04, duration: 0.8 });
        }
      }

      // Boost focused nucleus; focused stars are driven by tick loop (full opacity)
      gsap.to((nucleus.material as THREE.MeshBasicMaterial), { opacity: 1.0, duration: 0.6 });
      for (const star of objects.starMeshes) {
        if (star.userData.starOrbit?.nucleusKey === key) {
          gsap.to((star.material as THREE.MeshBasicMaterial), { opacity: 1.0, duration: 0.8 });
        }
      }
    }
    ```

    ```typescript
    export function unfocusSphere(
      objects: SceneObjects,
      scene: THREE.Scene,
      _camera: THREE.Camera,
      _controls?: CameraControls,
    ): void {
      const prevKey = objects.focusedKey;
      objects.mode = 'global';
      objects.focusedKey = null;
      objects.currentRingDimension = null;
      objects.highlightedRingValue = null;
      clearFilter(objects);

      // Dispose ring lines (ORB-04 cleanup; prevents accumulation on cycling)
      disposeRingLines(objects.ringLineMeshes, scene);
      objects.ringLineMeshes = [];
      objects.ringLineValues = [];

      // Clear ringOrbit from all stars (tick loop returns to Rodrigues)
      for (const star of objects.starMeshes) {
        star.userData.ringOrbit = undefined;
        star.userData.isStagingTarget = false; // tick takes over star positions
      }

      // Contraction: return non-focused nuclei to homePos (ORB-07)
      for (const [k, mesh] of objects.nucleusMeshes.entries()) {
        if (k === prevKey) continue; // the previously focused nucleus doesn't need contraction
        const home = mesh.userData.homePos as THREE.Vector3;
        if (!home) continue;

        gsap.to(mesh.position, { x: home.x, y: home.y, z: home.z,
          duration: 1.4, ease: 'power2.inOut' });
        if (mesh.userData.halo) {
          gsap.to((mesh.userData.halo as THREE.Mesh).position,
            { x: home.x, y: home.y, z: home.z, duration: 1.4, ease: 'power2.inOut' });
        }
        // Stars clear isStagingTarget above — tick loop lerps them back to Rodrigues positions
      }

      // Restore all nuclei and star opacities
      for (const mesh of objects.nucleusMeshes.values()) {
        gsap.to((mesh.material as THREE.MeshBasicMaterial), { opacity: 0.9, duration: 0.8 });
        if (mesh.userData.halo) {
          gsap.to(((mesh.userData.halo as THREE.Mesh).material as THREE.MeshBasicMaterial),
            { opacity: 0.06, duration: 0.8 });
        }
      }
      for (const star of objects.starMeshes) {
        gsap.to((star.material as THREE.MeshBasicMaterial), { opacity: 0.85, duration: 0.8 });
      }
    }
    ```
