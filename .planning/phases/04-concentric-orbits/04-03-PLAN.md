    **Step 2: Add switchRingDimension to scene-state.ts**

    Export this function from scene-state.ts (it needs direct access to SceneObjects and needs scene to add ring lines):

    ```typescript
    export function switchRingDimension(
      objects: SceneObjects,
      scene: THREE.Scene,
      newDimension: RingDimension,
      config: ConcentricConfig,
    ): void {
      if (!objects.focusedKey || objects.currentRingDimension === newDimension) return;

      const key = objects.focusedKey;
      const nucleusOrbit = objects.nucleusOrbits[key];
      const nucleus = objects.nucleusMeshes.get(key)!;
      const color = (nucleus.material as THREE.MeshBasicMaterial).color.getHex();
      const enrichedCases = nucleusOrbit.cases as EnrichedCaseStudy[];

      // Step 1: Dispose old ring lines
      disposeRingLines(objects.ringLineMeshes, scene);
      objects.ringLineMeshes = [];
      objects.ringLineValues = [];
      objects.highlightedRingValue = null; // clear any highlight

      // Step 2: Recompute ring orbits with new dimension
      const ringOrbits = computeConcentricOrbits(enrichedCases, newDimension, nucleusOrbit, config);
      objects.currentRingDimension = newDimension;

      // Step 3: Update star.userData.ringOrbit
      for (const star of objects.starMeshes) {
        if (star.userData.starOrbit?.nucleusKey !== key) continue;
        const caseId = (star.userData.caseStudy as CaseStudy).id;
        const ro = ringOrbits.find(r => r.caseStudy.id === caseId);
        if (ro) star.userData.ringOrbit = ro;
      }

      // Step 4: Create new ring lines
      const nucCenter = new THREE.Vector3(nucleus.position.x, nucleus.position.y, nucleus.position.z);
      const u = new THREE.Vector3(nucleusOrbit.orbitU.x, nucleusOrbit.orbitU.y, nucleusOrbit.orbitU.z);
      const v = new THREE.Vector3(nucleusOrbit.orbitV.x, nucleusOrbit.orbitV.y, nucleusOrbit.orbitV.z);
      const planeNormal = new THREE.Vector3().crossVectors(u, v).normalize();

      const seenRadii = new Map<string, number>();
      for (const ro of ringOrbits) seenRadii.set(ro.ringValue, ro.ringRadius);

      for (const [ringValue, radius] of seenRadii.entries()) {
        const line = buildRingLine(radius, nucCenter, planeNormal, color);
        scene.add(line);
        objects.ringLineMeshes.push(line);
        objects.ringLineValues.push(ringValue);
      }
      // Stars glide to new positions automatically via tick loop lerp (0.08 factor)
    }
    ```

    At the end of the enterFocus function, wire up the switcher:
    ```typescript
    // Ring dimension switcher (ORB-03)
    const switcherEl = document.getElementById('ring-switcher')!;
    // Populate buttons
    switcherEl.innerHTML = ALL_DIMS
      .map(d => `<button data-dim="${d}" class="${d === sceneObjects.currentRingDimension ? 'active' : ''}">${DIM_LABELS[d]}</button>`)
      .join('');

    // AbortController so we can remove the listener cleanly in exitFocus
    const switcherAbort = new AbortController();
    switcherEl.addEventListener('click', (e) => {
      const btn = (e.target as HTMLElement).closest<HTMLButtonElement>('[data-dim]');
      if (!btn) return;
      const newDim = btn.getAttribute('data-dim') as RingDimension;
      if (newDim === sceneObjects.currentRingDimension) return;

      const concentricConfig = {
        baseRadius: 40,
        ringSpacing: 25, // simplified constant for switcher calls
        angularSpeedBase: 0.00003,
      };
      switchRingDimension(sceneObjects, scene, newDim, concentricConfig);

      // Update active button state
      switcherEl.querySelectorAll<HTMLButtonElement>('[data-dim]').forEach(b => {
        b.classList.toggle('active', b.getAttribute('data-dim') === newDim);
      });
    }, { signal: switcherAbort.signal });

    // Store abort controller for cleanup in exitFocus
    (sceneObjects as any)._switcherAbort = switcherAbort;
    ```

    In exitFocus, before other cleanup:
    ```typescript
    // Clean up ring switcher listener
    const abort = (sceneObjects as any)._switcherAbort as AbortController | undefined;
    abort?.abort();
    delete (sceneObjects as any)._switcherAbort;
    // Clear switcher buttons
    const switcherEl = document.getElementById('ring-switcher');
    if (switcherEl) switcherEl.innerHTML = '';
    ```
